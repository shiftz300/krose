function(krose_test name source)
  add_executable(test-${name} ${source})
  target_compile_options(test-${name} PRIVATE
    $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>
    $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-UNDEBUG>)
  set_target_properties(test-${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  add_test(NAME ${name} COMMAND test-${name} ${ARGN})
endfunction()
function(krose_core_test name source)
  krose_test(${name} ${source} ${ARGN})
  target_link_libraries(test-${name} PRIVATE krose::core)
endfunction()
