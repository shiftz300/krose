# Optional design tool. Never included by the plugin/core build.
cmake_minimum_required(VERSION 3.25)
project(KroseDesignPreview LANGUAGES NONE)
get_filename_component(KROSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(KROSE_FACTORY_ASSET_DIR "${KROSE_ROOT}/assets/program-library" CACHE PATH "Frozen program catalog")
set(_editor_web "${CMAKE_CURRENT_SOURCE_DIR}/reference")
function(krose_read_editor file variable)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" _content)
  set(${variable} "${_content}" PARENT_SCOPE)
endfunction()
krose_read_editor("${_editor_web}/font.css" KROSE_FONT)
krose_read_editor("${_editor_web}/panel.css" KROSE_CSS)
krose_read_editor("${_editor_web}/panel.html" KROSE_MARKUP)
krose_read_editor("${_editor_web}/panel.js" KROSE_PANEL_JS)
krose_read_editor("${_editor_web}/preview-bridge.js" KROSE_BRIDGE)
krose_read_editor("${_editor_web}/control-bindings.js" KROSE_BINDINGS)
krose_read_editor("${_editor_web}/preset-browser.js" KROSE_BROWSER_SCRIPT)
krose_read_editor("${KROSE_FACTORY_ASSET_DIR}/FactoryPrograms.json" KROSE_CATALOG)
krose_read_editor("${KROSE_FACTORY_ASSET_DIR}/FactoryBrowser.json" KROSE_BROWSER)
configure_file("${_editor_web}/preview.html.in" "${CMAKE_CURRENT_BINARY_DIR}/preview.html" @ONLY)
